cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(openpack_fw)
target_sources(app PRIVATE src/main.c)
target_include_directories(app PRIVATE include src/proto src/td)
target_sources(app PRIVATE src/td/op_td.c) # AIR-3 / RX-4 time diversity (pack app core, receiver net core)
target_sources(app PRIVATE src/proto/proto.c src/proto/proto_rx.c src/proto/proto_pack.c)
if(CONFIG_OP_NRF5340 AND NOT CONFIG_SOC_COMPATIBLE_NRF5340_CPUNET)
  target_sources(app PRIVATE src/proto/proto_uart.c)
endif()
target_sources_ifdef(CONFIG_OP_NRF5340 app PRIVATE src/ipc.c)
if(CONFIG_OP_FEM_NRF21540 AND CONFIG_SOC_COMPATIBLE_NRF5340_CPUNET)
  target_sources(app PRIVATE src/net/op_fem.c)
endif()
if(CONFIG_OP_NRF5340 AND CONFIG_OP_ROLE_PACK)
  if(CONFIG_SOC_COMPATIBLE_NRF5340_CPUNET)
    target_sources(app PRIVATE src/net/sched.c)
  else()
    target_sources(app PRIVATE src/app/pack.c codec/adpcm6.c)
    target_include_directories(app PRIVATE codec)
    if(CONFIG_OP_TEST_APP_LOAD_PERIOD_US GREATER 0)
      target_sources(app PRIVATE src/app/testload.c)
    endif()
  endif()
endif()
if(CONFIG_OP_TEST_BSIM_ARGS)
  target_sources(app PRIVATE src/bsim_args.c)
  target_include_directories(app PRIVATE $ENV{BSIM_COMPONENTS_PATH}/libUtilv1/src)
endif()
include(src/rx/rx.cmake)
include(src/audio/audio.cmake)
